% ============ HEADINGS ============
% Chapter headings as "V. Tiêu đề" on one line. The chapter counter prints in
% Roman numerals, while sections, figures, tables and equations keep an arabic
% chapter prefix (hình 5.2, mục 5.1).
\usepackage{titlesec}

\renewcommand{\thechapter}{\Roman{chapter}}
\renewcommand{\thesection}{\arabic{chapter}.\arabic{section}}
\renewcommand{\thefigure}{\arabic{chapter}.\arabic{figure}}
\renewcommand{\thetable}{\arabic{chapter}.\arabic{table}}
\renewcommand{\theequation}{\arabic{chapter}.\arabic{equation}}

\titleformat{\chapter}[hang]{\normalfont\Large\bfseries}{\thechapter.}{0.5em}{}
\titlespacing*{\chapter}{0pt}{-12pt}{18pt}
\titleformat{\section}[hang]{\normalfont\large\bfseries}{\thesection}{0.6em}{}
\titleformat{\subsection}[hang]{\normalfont\normalsize\bfseries}{\thesubsection}{0.6em}{}

% Appendix chapters are lettered, and so are their floats (bảng A.1).
\let\layoutappendix\appendix
\renewcommand{\appendix}{%
  \layoutappendix
  \renewcommand{\thechapter}{\Alph{chapter}}%
  \renewcommand{\thesection}{\Alph{chapter}.\arabic{section}}%
  \renewcommand{\thetable}{\Alph{chapter}.\arabic{table}}%
  \renewcommand{\thefigure}{\Alph{chapter}.\arabic{figure}}%
  \renewcommand{\theequation}{\Alph{chapter}.\arabic{equation}}%
}

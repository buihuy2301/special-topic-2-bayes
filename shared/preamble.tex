% Figure inclusion that tolerates figures which have not been generated yet.
% Figures come from \figdir as PNG, written by the report's own script/notebook.
% [#1] = width as a fraction of \linewidth, #2 = file stem, #3 = caption,
% #4 = label
% ---------------------------------------------------------------------------
% The figure stems contain underscores, so the name has to be detokenized
% before it is typeset.
\newcommand{\missingfigbox}[1]{%
  \fbox{\parbox[c][3.2cm][c]{0.86\linewidth}{\centering
    \texttt{\detokenize{#1}.png}\par\vspace{4pt}
    \small chưa được sinh ra từ notebook\par}}%
}

% Placement is [tp] rather than [htbp] on purpose: with 'h' a figure declared
% between two paragraphs cuts the argument in half, and with 'b' it lands under
% a paragraph that continues on the next page.
\newcommand{\resultfig}[4][0.8]{%
  \begin{figure}[tp]
    \centering
    \IfFileExists{\figdir/#2.png}
      {\includegraphics[width=#1\linewidth]{\figdir/#2.png}}
      {\missingfigbox{#2}}
    \caption{#3}
    \label{#4}
  \end{figure}%
}
